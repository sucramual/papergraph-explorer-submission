# PaperGraph Explorer

**Interactive knowledge graph for academic papers with natural language queries**

Built for AI-Memory Hackathon (2 hour demo target)

---

## 🎯 Core Features

1. **Interactive Graph Visualization** - Click any paper node to select it
2. **Natural Language Queries** - Ask questions about selected paper:
   - "What are the most relevant papers to this?"
   - "Of all papers this cites, which is most prominent?"
   - "Who are the main authors in this research area?"

---

## 🏗️ Architecture

```
arXiv API (50 papers) → Download metadata (authors, citations)
    ↓
cognee.add() + cognee.cognify() → Build knowledge graph
    ↓
Qdrant (6 collections) → Store vectors + graph
    ↓
FastAPI Backend → Serve graph data + handle queries
    ↓
Interactive UI (vis.js) → Render clickable graph
    ↓
OpenAI GPT-4 → Answer queries about selected paper
```

### 🔨 Step 2: Ingest Papers + Build Graph

**File:** `ingest_papers.py`

```python
"""
Ingest papers into cognee and build knowledge graph.
"""
import os
import json
from dotenv import load_dotenv

load_dotenv()

# Configure cognee BEFORE importing
os.environ["VECTOR_DB_PROVIDER"] = "qdrant"
os.environ["VECTOR_DB_URL"] = os.getenv("QDRANT_URL", "http://localhost:6333")
os.environ["VECTOR_DB_KEY"] = os.getenv("QDRANT_API_KEY", "")

os.environ["LLM_PROVIDER"] = "openai"
os.environ["LLM_MODEL"] = "gpt-4o-mini"
os.environ["LLM_API_KEY"] = os.getenv("OPENAI_API_KEY")

os.environ["EMBEDDING_PROVIDER"] = "ollama"
os.environ["EMBEDDING_MODEL"] = "nomic-embed-text:latest"
os.environ["EMBEDDING_ENDPOINT"] = "http://localhost:11434/api/embed"
os.environ["EMBEDDING_DIMENSIONS"] = "768"

import cognee_community_vector_adapter_qdrant.register
import cognee
import asyncio

async def main():
    # Clear old data
    await cognee.prune.prune_data()
    await cognee.prune.prune_system(metadata=True)

    # Load papers metadata
    with open("papers_metadata.json") as f:
        papers = json.load(f)

    print(f"Ingesting {len(papers)} papers...")

    # Add each paper to cognee
    for i, paper in enumerate(papers, 1):
        text = f"""
Paper: {paper['title']}
Authors: {', '.join(paper['authors'])}
arXiv ID: {paper['id']}
Published: {paper['published']}

Abstract:
{paper['summary']}
"""
        await cognee.add(text, dataset_name="papers")
        print(f"Added {i}/{len(papers)}: {paper['title'][:50]}...")

    # Build knowledge graph
    print("\nBuilding knowledge graph...")
    await cognee.cognify()

    print("✓ Graph built successfully!")

if __name__ == "__main__":
    asyncio.run(main())
```

**What cognee extracts:**
- Papers (entities)
- Authors (entities)
- Concepts (from abstracts: "attention", "transformer", etc.)
- Relationships: "authored_by", "about"

```python
# Serve static files
app.mount("/static", StaticFiles(directory="static"), name="static")

@app.get("/")

    return FileResponse("static/index.html")

@app.get("/api/graph")
async def get_graph():
    """
    Get graph data for visualization.
    Returns: {nodes: [...], edges: [...]}
    """
    # TODO: Query cognee graph and convert to vis.js format
    # For now, return sample data
    return {
        "nodes": [],
        "edges": []
    }

class QueryRequest(BaseModel):
    paper_id: str
    question: str

@app.post("/api/query")
async def query_paper(req: QueryRequest):
    """
    Answer question about a selected paper.
    """
    # Use custom retriever to get graph context + OpenAI
    system_prompt_path = pathlib.Path("prompts/system_prompt.txt").resolve()

    retriever = GraphCompletionRetrieverWithUserPrompt(
        user_prompt_filename="user_prompt.txt",
        system_prompt_path=str(system_prompt_path),
        top_k=10,
    )

    # Build query with paper context
    query = f"Paper ID: {req.paper_id}\nQuestion: {req.question}"

    completion = await retriever.get_completion(query=query)

    return {"answer": completion[0]}

if __name__ == "__main__":
    import uvicorn
    uvicorn.run(app, host="0.0.0.0", port=7777)
```

#### 1. **Metadata Parser Missing** 🔴 HIGH PRIORITY
**Problem:** Download script outputs `.txt`, but app needs `.json`

**Solution:** Create `parse_metadata.py`
```python
import re
import json

def parse_metadata_txt(filepath):
    """Parse papers_metadata.txt to JSON."""
    with open(filepath) as f:
        content = f.read()

    # Split by paper separator
    paper_blocks = content.split('='*80)[1:]  # Skip empty first

    papers = []
    for block in paper_blocks:
        if not block.strip():
            continue

        # Extract fields with regex
        paper = {}

        id_match = re.search(r'ID: (.+)', block)
        if id_match:
            paper['id'] = id_match.group(1).strip()

        title_match = re.search(r'Title: (.+)', block)
        if title_match:
            paper['title'] = title_match.group(1).strip()

        authors_match = re.search(r'Authors: (.+)', block)
        if authors_match:
            paper['authors'] = [a.strip() for a in authors_match.group(1).split(',')]

        pub_match = re.search(r'Published: (.+)', block)
        if pub_match:
            paper['published'] = pub_match.group(1).strip()

        pdf_match = re.search(r'PDF URL: (.+)', block)
        if pdf_match:
            paper['pdf_url'] = pdf_match.group(1).strip()

        abstract_match = re.search(r'Abstract:\n(.+)', block, re.DOTALL)
        if abstract_match:
            paper['summary'] = abstract_match.group(1).strip()

        if paper:
            papers.append(paper)

    return papers

if __name__ == "__main__":
    papers = parse_metadata_txt("data/papers_metadata.txt")
    with open("papers_metadata.json", "w") as f:
        json.dump(papers, f, indent=2)
    print(f"✓ Parsed {len(papers)} papers to papers_metadata.json")
```

#### 2. **Graph Extraction Logic Missing** 🔴 HIGH PRIORITY
**Problem:** `/api/graph` endpoint returns empty data. Need to query cognee graph and convert to vis.js format.

**Solution:** Add graph extraction in `app.py`
```python
from cognee.infrastructure.databases.graph import get_graph_engine

async def extract_graph_data():
    """Extract graph from cognee and convert to vis.js format."""
    graph_engine = await get_graph_engine()

    # Query all paper entities
    query = """
    MATCH (p:Entity)
    WHERE p.entity_type = 'Paper' OR p.entity_type = 'Document'
    RETURN p
    """
    papers = await graph_engine.query(query)

    # Query relationships
    rel_query = """
    MATCH (a)-[r]->(b)
    WHERE a.entity_type = 'Paper' OR a.entity_type = 'Document'
    RETURN a, r, b
    """
    relationships = await graph_engine.query(rel_query)

    # Convert to vis.js format
    nodes = [{"id": p.id, "label": p.name[:50], "title": p.name} for p in papers]
    edges = [{"from": r.source, "to": r.target, "label": r.type} for r in relationships]

    return {"nodes": nodes, "edges": edges}
```

**Fallback Plan:** If cognee graph is sparse, build graph directly from papers_metadata.json:
```python
# papers[0-9] all relate to papers[10-49] (search results are "related")
```
